  res_surj : ∀ x : R, ∃ a : O, x - algebraMap O R a ∈ maximalIdeal R

theorem resMap_bijective : Function.Bijective (ResidueField.map (algebraMap O R)) := by
  refine ⟨(ResidueField.map (algebraMap O R)).injective, fun y => ?_⟩
  obtain ⟨x, rfl⟩ := residue_surjective y
  obtain ⟨a, ha⟩ := IsCO.res_surj (O := O) x
  refine ⟨residue O a, ?_⟩
  rw [ResidueField.map_residue]
  exact ((Ideal.Quotient.eq).2 ha).symm

def res : R →ₐ[O] ResidueField O :=
  { ((RingEquiv.ofBijective _ (resMap_bijective O R)).symm.toRingHom.comp (residue R)) with
    commutes' := fun a => by
      show (RingEquiv.ofBijective _ (resMap_bijective O R)).symm (residue R (algebraMap O R a)) = residue O a
      rw [RingEquiv.symm_apply_eq]
      exact (ResidueField.map_residue _ _).symm }

theorem res_eq_zero_iff (x : R) : res O R x = 0 ↔ x ∈ maximalIdeal R := by
  show (RingEquiv.ofBijective _ (resMap_bijective O R)).symm (residue R x) = 0 ↔ _
  rw [map_eq_zero_iff _ (RingEquiv.injective _), residue_eq_zero_iff]

theorem res_map {R' : Type} [CommRing R'] [Algebra O R'] [IsCO O R'] (f : R →ₐ[O] R') (x : R) :
    res O R' (f x) = res O R x := by
  obtain ⟨a, ha⟩ := IsCO.res_surj (O := O) x
  have h1 : res O R x = algebraMap O _ a := by
    have := (res_eq_zero_iff O R _).2 ha
    rwa [map_sub, AlgHom.commutes, sub_eq_zero] at this
  have hm : ∀ y ∈ maximalIdeal R, f y ∈ maximalIdeal R' := fun y hy => by
    obtain ⟨b, hb⟩ := IsCO.res_surj (O := O) (f y)
    by_cases hu : IsUnit b
    · exfalso
      have h2 : IsUnit (y - algebraMap O R b) := by
        by_contra hn
        have h3 : y - algebraMap O R b ∈ maximalIdeal R := hn
        have h4 : algebraMap O R b ∈ maximalIdeal R := by simpa using Ideal.sub_mem _ hy h3
        exact h4 (hu.map _)
      have h5 := h2.map f
      rw [map_sub, AlgHom.commutes] at h5
      exact hb h5
    · have h6 : algebraMap O R' b ∈ maximalIdeal R' := fun h => hu (IsLocalHom.map_nonunit b h)
      simpa using Ideal.add_mem _ hb h6
  have h2 := (res_eq_zero_iff O R' _).2 (hm _ ha)
  rw [map_sub, map_sub, AlgHom.commutes, AlgHom.commutes, sub_eq_zero] at h2
  rw [h2, h1]

end Res

  rw [coeff_mul]
  refine Finset.sum_eq_zero fun p hp => ?_
  by_cases h1 : Finsupp.weight wt p.1 = i
  · have h2 : Finsupp.weight wt p.2 ≠ j := fun h2 => hn (by
      rw [← Finset.HasAntidiagonal.mem_antidiagonal.1 hp, map_add, h1, h2])

    Matrix (Fin n) (Fin n) K := (List.ofFn fun i => r (g i) - ψ i (g i) • (1 : Matrix (Fin n) (Fin n) K)).prod

theorem ordProd_map {H K K' : Type} [CommRing K] [CommRing K'] {n : ℕ} (f : K →+* K')

  refine congrArg List.ofFn (funext fun i => ?_)
  simp only [Function.comp_apply, map_sub]

@[ext] structure NRep (K : Type) [CommRing K] where
  r : F.G0 →* Matrix (Fin n) (Fin n) K
  ψ : ∀ v, Fin n → F.Gv v →* Kˣ
  pol : ∀ g, r (F.star g) = Jmat K n * (r g)ᵀ * Jmat K n
  diag : ∀ i j, i ≠ j → (r F.g0 - r (F.star F.g0)) i j = 0
  ord : ∀ v g, ordProd (fun s => r (F.loc0 v s)) (fun i s => (ψ v i s : K)) g = 0

def map (f : K →+* K') (ρ : NRep F n K) : NRep F n K' where
  r := (f.mapMatrix : Matrix (Fin n) (Fin n) K →+* _).toMonoidHom.comp ρ.r
  ψ v i := (Units.map f.toMonoidHom).comp (ρ.ψ v i)
  pol g := by
    show f.mapMatrix (ρ.r (F.star g)) = Jmat K' n * (f.mapMatrix (ρ.r g))ᵀ * Jmat K' n
    rw [ρ.pol, map_mul, map_mul]
    simp only [RingHom.mapMatrix_apply, Jmat_map, Matrix.transpose_map]
  diag i j h := by
    show f (ρ.r F.g0 i j) - f (ρ.r (F.star F.g0) i j) = 0
    rw [← map_sub]
    exact (congrArg f (ρ.diag i j h)).trans (map_zero f)
  ord v g := (ordProd_map f _ _ g).trans (by rw [ρ.ord, map_zero])

def restrict {O : Type} [CommRing O] [Algebra O K] (ρ : NRep F n K) (A : Subalgebra O K)
    (h1 : ∀ g i j, ρ.r g i j ∈ A) (h2 : ∀ v i g, (ρ.ψ v i g : K) ∈ A) : NRep F n A where
  r :=
    { toFun := fun g => Matrix.of fun i j => ⟨ρ.r g i j, h1 g i j⟩
      map_one' := by
        ext i j
        by_cases h : i = j <;> simp [h, Matrix.one_apply]
      map_mul' := fun g h => by
        ext i j
        simp [Matrix.mul_apply] }
  ψ v i :=
    { toFun := fun g => ⟨⟨ρ.ψ v i g, h2 v i g⟩, ⟨ρ.ψ v i g⁻¹, h2 v i g⁻¹⟩, Subtype.ext (by simp), Subtype.ext (by simp)⟩
      map_one' := by ext; simp
      map_mul' := fun g h => by ext; simp }
  pol g := by
    refine Matrix.map_injective (f := (A.val : A →+* K)) Subtype.val_injective ?_
    show _ = (A.val : A →+* K).mapMatrix _
    rw [map_mul, map_mul]
    simp only [RingHom.mapMatrix_apply, Jmat_map]
    exact ρ.pol g
  diag i j h := Subtype.ext (ρ.diag i j h)
  ord v g := by
    refine Matrix.map_injective (f := (A.val : A →+* K)) Subtype.val_injective ?_
    show (A.val : A →+* K).mapMatrix _ = (A.val : A →+* K).mapMatrix _

def Cont (I : Ideal K) (ρ : NRep F n K) : Prop :=
  (∀ i j, AdicCont I fun g => ρ.r g i j) ∧ ∀ v i, AdicCont I fun g => (ρ.ψ v i g : K)

def IsBar (f : k →+* K) (ρ : NRep F n K) : Prop :=
  (∀ g i, ρ.r g i i = f ((F.Ψ g ^ w n i : kˣ) : k)) ∧ ∀ v i g, (ρ.ψ v i g : K) = f ((F.Ψ (F.loc v g) ^ w n i : kˣ) : k)

def IsDiag (ρ : NRep F n K) : Prop := ∀ g i j, i ≠ j → ρ.r g i j = 0

def triP (ρ : NRep F n K) : Ideal K := Ideal.span {x | ∃ g i j, j < i ∧ x = ρ.r g i j}

def triM (ρ : NRep F n K) : Ideal K := Ideal.span {x | ∃ g i j, i < j ∧ x = ρ.r g i j}

def ee (ρ : NRep F n K) : Ideal K := Ideal.span {x | ∃ g i j, Odd (w n i - w n j) ∧ x = ρ.r g i j}

def adicClosure {O R : Type} [CommRing O] [CommRing R] [IsLocalRing R] [Algebra O R] (S : Subalgebra O R) :
    Subalgebra O R where
  carrier := {x | ∀ N : ℕ, ∃ y ∈ S, x - y ∈ maximalIdeal R ^ N}
  mul_mem' {a b} ha hb N := by
    obtain ⟨x, hx, h1⟩ := ha N
    obtain ⟨y, hy, h2⟩ := hb N
    refine ⟨x * y, mul_mem hx hy, ?_⟩
    have : a * b - x * y = (a - x) * b + x * (b - y) := by ring
    rw [this]
    exact add_mem (Ideal.mul_mem_right _ _ h1) (Ideal.mul_mem_left _ _ h2)
  add_mem' {a b} ha hb N := by
    obtain ⟨x, hx, h1⟩ := ha N
    obtain ⟨y, hy, h2⟩ := hb N
    refine ⟨x + y, add_mem hx hy, ?_⟩
    rw [add_sub_add_comm]
    exact add_mem h1 h2
  algebraMap_mem' c N := ⟨_, S.algebraMap_mem c, by simp⟩

theorem le_adicClosure {O R : Type} [CommRing O] [CommRing R] [IsLocalRing R] [Algebra O R] (S : Subalgebra O R) :

  have hspan : maximalIdeal R ≤ Ideal.span (Set.range e) ⊔ (maximalIdeal O).map (Algebra.ofId O R) ⊔

      have := hU.map π
      rw [hπX] at this
      exact hem _ this
    · rw [Ideal.map_le_iff_le_comap]

      refine Ideal.mem_map_of_mem π fun hU => ?_
      have := hU.map (constantCoeff : MvPowerSeries T O →+* O)
      rw [constantCoeff_C] at this
      exact ha this
  have hπm : ∀ i, π (MvPowerSeries.X i) ∈ maximalIdeal R := fun i => by rw [hπX]; exact hem _
  have hhom : IsHomog O (fun i : T => wt (u i)) (RingHom.ker π) := by
    refine homog_of_stable O X R T _ π hπm ?_

    let g : R →ₐ[O] R' := (Ideal.Quotient.liftₐ (RingHom.ker π) f hf).comp
      (Ideal.quotientKerAlgEquivOfSurjective hsurj).symm.toAlgHom
    have hg : ∀ y, g (π y) = f y := fun y => by
      have : (Ideal.quotientKerAlgEquivOfSurjective hsurj).symm (π y) = Ideal.Quotient.mk _ y := by
        rw [AlgEquiv.symm_apply_eq]; rfl
      show Ideal.Quotient.liftₐ (RingHom.ker π) f hf ((Ideal.quotientKerAlgEquivOfSurjective hsurj).symm (π y)) = _
      rw [this]
      rfl
    obtain ⟨g', hg'⟩ := hst R' g t ht
    have : g'.comp π = f' := ext_X _ _ (fun i => map_mem_max g' (hπm i)) fun i => by
      rw [AlgHom.comp_apply, hf', ← hg, hπX, hg']
    rw [← this, AlgHom.comp_apply, hx, map_zero]
  let W : WPres O X R := { σ := T, wt := fun i => wt (u i), π := π, surj := hsurj, homog := hhom }
  refine ⟨W, u, fun i => ⟨hπX i, rfl⟩, fun s => eigen_mem O X R W hπm (e s) (wt s) ?_⟩
  intro R' _ _ _ f f' t ht h
  obtain ⟨g', hg'⟩ := hst R' f t ht

    have h2 : f' (π (MvPowerSeries.X i)) = t (Multiplicative.ofAdd (wt (u i))) * f (π (MvPowerSeries.X i)) := h i
    rw [AlgHom.comp_apply, AlgHom.comp_apply, h2, hπX, hg']
  have h3 : f' (e s) = g' (e s) := by
    obtain ⟨y, hy⟩ := hsurj (e s)
    rw [← hy]
    exact (DFunLike.congr_fun h1 y).symm
  rw [h3, hg']

private theorem cont_map {O Γ R R' : Type} [CommRing O] [IsLocalRing O] [TopologicalSpace Γ] [CommRing R] [Algebra O R]
    [IsCO O R] [CommRing R'] [Algebra O R'] [IsCO O R'] (f : R →ₐ[O] R') (c : R') {a : Γ → R}
    (ha : AdicCont (maximalIdeal R) a) : AdicCont (maximalIdeal R') fun g => c * f (a g) := by
  have hm : Ideal.map f.toRingHom (maximalIdeal R) ≤ maximalIdeal R' := by
    rw [Ideal.map_le_iff_le_comap]
    intro x hx
    show f x ∈ maximalIdeal R'
    rw [← res_eq_zero_iff O R', res_map, res_eq_zero_iff]
    exact hx
  intro N
  have hN : maximalIdeal R ^ N ≤ Ideal.comap f.toRingHom (maximalIdeal R' ^ N) := by
    rw [← Ideal.map_le_iff_le_comap, Ideal.map_pow]
    exact Ideal.pow_right_mono hm N
  have e : (fun g => Ideal.Quotient.mk (maximalIdeal R' ^ N) (c * f (a g))) =
      (fun q => Ideal.Quotient.mk (maximalIdeal R' ^ N) c *
        Ideal.quotientMap (maximalIdeal R' ^ N) f.toRingHom hN q) ∘
        (fun g => Ideal.Quotient.mk (maximalIdeal R ^ N) (a g)) := by
    funext g
    simp [Ideal.quotientMap_mk]
  rw [e]
  exact (ha N).comp _

private theorem twist_lemma {l : ℕ} [Fact l.Prime] {O : Type} [CommRing O] [IsLocalRing O] {F : Frame l (ResidueField O)}
    {n : ℕ} (S : Space O F n) (C : Type) [CommRing C] [Algebra O C] [IsCO O C] (f : S.R →ₐ[O] C)
    (c : Fin n → Fin n → C) (hc : ∀ i, c i i = 1) (ρ' : NRep F n C)
    (hr : ∀ g i j, ρ'.r g i j = c i j * f (S.ρ.r g i j)) (hψ : ∀ v i g, (ρ'.ψ v i g : C) = f (S.ρ.ψ v i g)) :
    IsLift O ρ' := by
  obtain ⟨⟨hc1, hc2⟩, ⟨hb1, hb2⟩, hd⟩ := S.lift
  refine ⟨⟨fun i j => ?_, fun v i => ?_⟩, ⟨fun g i => ?_, fun v i g => ?_⟩, fun g i j hij => ?_⟩
  · simpa only [hr] using cont_map f (c i j) (hc1 i j)
  · simpa only [hψ, one_mul] using cont_map f 1 (hc2 v i)
  · show (res O C) (ρ'.r g i i) = _
    rw [hr, hc, one_mul, res_map]
    exact hb1 g i
  · show (res O C) ((ρ'.ψ v i g : Cˣ) : C) = _
    rw [hψ, res_map]
    exact hb2 v i g
  · show (res O C) (ρ'.r g i j) = 0
    rw [hr, map_mul, res_map]
    have h0 : (res O S.R) (S.ρ.r g i j) = 0 := hd g i j hij
    rw [h0, mul_zero]

private theorem twist {l : ℕ} [Fact l.Prime] {O : Type} [CommRing O] [IsLocalRing O] {F : Frame l (ResidueField O)}
    {n : ℕ} (S : Space O F n) (C : Type) [CommRing C] [Algebra O C] [IsCO O C] (f : S.R →ₐ[O] C) (u : Cˣ) :
    ∃ f' : S.R →ₐ[O] C, (∀ g i j, f' (S.ρ.r g i j) = ((u ^ (w n i - w n j) : Cˣ) : C) * f (S.ρ.r g i j)) ∧
      ∀ v i g, f' (S.ρ.ψ v i g) = f (S.ρ.ψ v i g) := by
  obtain ⟨ρ', hr, hψ⟩ := alg_twist u (S.ρ.map (f : S.R →+* C))
  have hl : IsLift O ρ' := twist_lemma S C f (fun i j => ((u ^ (w n i - w n j) : Cˣ) : C)) (by simp) ρ'
    (fun g i j => by rw [hr]; rfl) (fun v i g => by rw [hψ]; rfl)
  obtain ⟨f', hf', -⟩ := S.univ C ρ' hl
  refine ⟨f', fun g i j => ?_, fun v i g => ?_⟩
  · calc f' (S.ρ.r g i j) = (S.ρ.map (f' : S.R →+* C)).r g i j := rfl
      _ = ρ'.r g i j := by rw [hf']
      _ = _ := by rw [hr]; rfl
  · calc f' (S.ρ.ψ v i g : S.R) = (((S.ρ.map (f' : S.R →+* C)).ψ v i g : Cˣ) : C) := rfl
      _ = (ρ'.ψ v i g : C) := by rw [hf']
      _ = _ := by rw [hψ]; rfl

private theorem uniq {l : ℕ} [Fact l.Prime] {O : Type} [CommRing O] [IsLocalRing O] {F : Frame l (ResidueField O)}
    {n : ℕ} (S : Space O F n) (C : Type) [CommRing C] [Algebra O C] [IsCO O C] (f f' : S.R →ₐ[O] C)
    (h1 : ∀ g i j, f (S.ρ.r g i j) = f' (S.ρ.r g i j)) (h2 : ∀ v i g, f (S.ρ.ψ v i g) = f' (S.ρ.ψ v i g)) :
    f = f' := by
  have hl : IsLift O (S.ρ.map (f : S.R →+* C)) := twist_lemma S C f (fun _ _ => 1) (fun _ => rfl) _
    (fun g i j => by rw [one_mul]; rfl) (fun v i g => rfl)
  refine (S.univ C _ hl).unique rfl (NRep.ext ?_ ?_)
  · exact MonoidHom.ext fun g => Matrix.ext fun i j => (h1 g i j).symm
  · funext v i
    exact MonoidHom.ext fun g => Units.ext (h2 v i g).symm

private theorem coord {l : ℕ} [Fact l.Prime] {O : Type} [CommRing O] [IsLocalRing O] [IsNoetherianRing O]
    [IsAdicComplete (maximalIdeal O) O] {F : Frame l (ResidueField O)} {n : ℕ} (S : Space O F n) :
    ∃ W : WPres O ℤ S.R, (∀ s, W.z s ∈ maximalIdeal S.R) ∧ (∀ g i j, S.ρ.r g i j ∈ W.part (w n i - w n j)) ∧
      (∀ v i g, (S.ρ.ψ v i g : S.R) ∈ W.part 0) ∧
      ∀ s, (∃ g i j, W.z s = S.ρ.r g i j ∧ W.wt s = w n i - w n j) ∨
        (W.wt s = 0 ∧ W.z s ∈ Algebra.adjoin O (S.ρ.mon 0 ∪ {x | ∃ v i g, x = (S.ρ.ψ v i g : S.R)})) := by
  classical
  let a : S.R → O := fun x => Classical.choose (IsCO.res_surj (O := O) x)
  have ha : ∀ x, x - algebraMap O S.R (a x) ∈ maximalIdeal S.R := fun x =>
    Classical.choose_spec (IsCO.res_surj (O := O) x)
  let e : (F.G0 × Fin n × Fin n) ⊕ (Σ v : F.V, Fin n × F.Gv v) → S.R := Sum.elim
    (fun p => if p.2.1 = p.2.2 then S.ρ.r p.1 p.2.1 p.2.2 - algebraMap O S.R (a (S.ρ.r p.1 p.2.1 p.2.2))
      else S.ρ.r p.1 p.2.1 p.2.2)
    (fun q => (S.ρ.ψ q.1 q.2.1 q.2.2 : S.R) - algebraMap O S.R (a (S.ρ.ψ q.1 q.2.1 q.2.2)))
  let wt : (F.G0 × Fin n × Fin n) ⊕ (Σ v : F.V, Fin n × F.Gv v) → ℤ := Sum.elim
    (fun p => w n p.2.1 - w n p.2.2) (fun _ => 0)
  have eon : ∀ g i, e (Sum.inl (g, i, i)) = S.ρ.r g i i - algebraMap O S.R (a (S.ρ.r g i i)) :=
    fun g i => if_pos rfl
  have eoff : ∀ g i j, i ≠ j → e (Sum.inl (g, i, j)) = S.ρ.r g i j := fun g i j h => if_neg h
  have eψ : ∀ v i g, e (Sum.inr ⟨v, i, g⟩) =
      (S.ρ.ψ v i g : S.R) - algebraMap O S.R (a (S.ρ.ψ v i g)) := fun v i g => rfl
  have hoff : ∀ g i j, i ≠ j → S.ρ.r g i j ∈ maximalIdeal S.R := fun g i j h =>
    (res_eq_zero_iff O S.R _).mp (by exact S.lift.2.2 g i j h)
  have he : ∀ s, e s ∈ maximalIdeal S.R := by
    rintro (⟨g, i, j⟩ | ⟨v, i, g⟩)
    · by_cases h : i = j
      · subst h
        rw [eon]
        exact ha _
      · rw [eoff g i j h]
        exact hoff g i j h
    · rw [eψ]
      exact ha _
  have huniq : ∀ (R' : Type) [CommRing R'] [Algebra O R'] [IsCO O R'] (f f' : S.R →ₐ[O] R'),
      (∀ s, f (e s) = f' (e s)) → f = f' := by
    intro R' _ _ _ f f' h
    apply uniq S R' f f'
    · intro g i j
      by_cases hij : i = j
      · subst hij
        have h1 := h (Sum.inl (g, i, i))
        rw [eon, map_sub, map_sub, AlgHom.commutes, AlgHom.commutes] at h1
        exact sub_left_injective h1
      · have h1 := h (Sum.inl (g, i, j))
        rw [eoff g i j hij] at h1
        exact h1
    · intro v i g
      have h1 := h (Sum.inr ⟨v, i, g⟩)
      rw [eψ, map_sub, map_sub, AlgHom.commutes, AlgHom.commutes] at h1
      exact sub_left_injective h1
  have htw : ∀ (R' : Type) [CommRing R'] [Algebra O R'] [IsCO O R'] (f : S.R →ₐ[O] R')
      (t : Multiplicative ℤ →* R'ˣ), (∀ j, res O R' (t j) = 1) →
      ∃ f' : S.R →ₐ[O] R', ∀ s, f' (e s) = (t (Multiplicative.ofAdd (wt s)) : R') * f (e s) := by
    intro R' _ _ _ f t _
    obtain ⟨f', h1, h2⟩ := twist S R' f (t (Multiplicative.ofAdd 1))
    have ht : ∀ k : ℤ, t (Multiplicative.ofAdd k) = t (Multiplicative.ofAdd 1) ^ k := by
      intro k
      rw [← map_zpow, ← ofAdd_zsmul, smul_eq_mul, mul_one]
    refine ⟨f', ?_⟩
    rintro (⟨g, i, j⟩ | ⟨v, i, g⟩)
    · by_cases hij : i = j
      · subst hij
        show f' (e (Sum.inl (g, i, i))) = (t (Multiplicative.ofAdd (w n i - w n i)) : R') * _
        rw [sub_self, eon, map_sub, map_sub, AlgHom.commutes, AlgHom.commutes, h1]
        simp
      · show f' (e (Sum.inl (g, i, j))) = (t (Multiplicative.ofAdd (w n i - w n j)) : R') * _
        rw [eoff g i j hij, h1, ht (w n i - w n j)]
    · show f' (e (Sum.inr ⟨v, i, g⟩)) = (t (Multiplicative.ofAdd 0) : R') * _
      rw [eψ, map_sub, map_sub, AlgHom.commutes, AlgHom.commutes, h2]
      simp
  obtain ⟨W, u, hWu, hpart⟩ := coord_grading O ℤ S.R _ e wt he huniq htw
  have h1mem : ∀ c : O, algebraMap O S.R c ∈ W.part 0 := fun c => by
    rw [Algebra.algebraMap_eq_smul_one]
    exact Submodule.smul_mem _ c SetLike.GradedOne.one_mem
  refine ⟨W, fun s => by rw [(hWu s).1]; exact he _, ?_, ?_, ?_⟩
  · intro g i j
    by_cases hij : i = j
    · subst hij
      have hp : S.ρ.r g i i - algebraMap O S.R (a (S.ρ.r g i i)) ∈ W.part 0 := by
        have := hpart (Sum.inl (g, i, i))

[RTC_SRC_PART_SPAN] Finite homogeneous monomial spanning
https://github.com/CocoSgt/riemann-hypothesis-lean/blob/7cbf7a0c32fca1badf05931879a37c3f8a2b03b4/src/Development/AutomorphicArithmetic/AutomorphicVarieties/CommutativeAlgebra/PartSpan.lean#L5-L51
theorem UpperTheory.V.Alg.part_span :
    ∀ (O R : Type) [CommRing O] [CommRing R] [Algebra O R] (X : Type) [AddCommGroup X] [Module.Free ℤ X] [Module.Finite ℤ X] (W : WPres O X R) (S : Set (W.σ →₀ ℕ)) (j : X), (∀ n ∈ S, Finsupp.weight W.wt n = j) →
      ∃ μ : Finset (W.σ →₀ ℕ), (μ : Set _) ⊆ S ∧ ∀ f : MvPowerSeries W.σ O, (∀ n, n ∉ S → coeff n f = 0) →
        ∃ b : (W.σ →₀ ℕ) → MvPowerSeries W.σ O, (∀ m, b m ∈ wtPart O W.wt 0) ∧ f = ∑ m ∈ μ, b m * monomial m 1 := by
  intro O R _ _ _ X _ _ _ W S j hS
  classical
  haveI := W.fin
  have hMfin : {m : W.σ →₀ ℕ | Minimal (· ∈ S) m}.Finite :=
    WellQuasiOrderedLE.finite_of_isAntichain (setOf_minimal_antichain _)
  have hex : ∀ n ∈ S, ∃ m ∈ {m : W.σ →₀ ℕ | Minimal (· ∈ S) m}, m ≤ n := by
    intro n hn
    obtain ⟨b, hb, hmin⟩ := (Set.isPWO_of_wellQuasiOrderedLE S).exists_le_minimal hn
    exact ⟨b, hmin, hb⟩
  choose! c hcM hcle using hex
  refine ⟨hMfin.toFinset, ?_, ?_⟩
  · rw [Set.Finite.coe_toFinset]
    exact fun m hm => hm.1
  · intro f hf
    obtain ⟨b, hb⟩ : ∃ b : (W.σ →₀ ℕ) → MvPowerSeries W.σ O, ∀ m k,
        coeff k (b m) = if k + m ∈ S ∧ c (k + m) = m then coeff (k + m) f else 0 :=
      ⟨fun m k => if k + m ∈ S ∧ c (k + m) = m then coeff (k + m) f else 0, fun _ _ => rfl⟩
    refine ⟨b, ?_, ?_⟩
    · intro m k hk
      rw [hb, if_neg]
      rintro ⟨h1, h2⟩
      apply hk
      have hm : Finsupp.weight W.wt m = j := hS m (by rw [← h2]; exact (hcM _ h1).1)
      have hkm := hS _ h1
      rw [map_add, hm] at hkm
      simpa using hkm
    · ext n
      rw [map_sum, Finset.sum_congr rfl
        (g := fun m => if c n = m then (if n ∈ S then coeff n f else 0) else 0), Finset.sum_ite_eq]
      · by_cases hn : n ∈ S
        · rw [if_pos ((Set.Finite.mem_toFinset hMfin).2 (hcM n hn)), if_pos hn]
        · rw [hf n hn]
          simp
      · intro m _
        rw [MvPowerSeries.coeff_mul_monomial, mul_one]
        by_cases hmn : m ≤ n
        · rw [if_pos hmn, hb, tsub_add_cancel_of_le hmn]
          by_cases h1 : n ∈ S <;> by_cases h2 : c n = m <;> simp [h1, h2]
        · rw [if_neg hmn]
          by_cases h1 : n ∈ S
          · have h3 : c n ≠ m := fun h => hmn (h ▸ hcle n h1)
            simp [h3]
          · simp [h1]

theorem UpperTheory.V.Alg.part_filt :
    ∀ (O : Type) [CommRing O] [IsLocalRing O] [IsNoetherianRing O] [IsAdicComplete (maximalIdeal O) O] (X : Type) [AddCommGroup X] [Module.Free ℤ X] [Module.Finite ℤ X] [DecidableEq X] (R : Type) [CommRing R] [Algebra O R] [IsCO O R] (W : WPres O X R), (∀ s, W.z s ∈ maximalIdeal R) → ∀ (j : X) (n : ℕ), ∃ M : ℕ, W.part j ⊓ (maximalIdeal R ^ M).restrictScalars O ≤ (W.part 0 ⊓ (maximalIdeal R).restrictScalars O) ^ n * W.part 0 * W.part j := by
  intro O _ _ _ _ X _ _ _ _ R _ _ _ W _ j k
  classical
  set Q : Submodule O R := W.part 0 ⊓ (maximalIdeal R).restrictScalars O with hQ
  have hloc : IsLocalHom (W.π : MvPowerSeries W.σ O →+* R) := IsLocalHom.of_surjective _ W.surj
  have hone : (1 : R) ∈ W.part 0 := ⟨1, one_mem_wtPart O W.wt, map_one W.π⟩

  have hmono : ∀ k : ℕ, ∃ N : ℕ, ∀ n : W.σ →₀ ℕ, Finsupp.weight W.wt n = j → N ≤ n.degree →
      W.π (monomial n 1) ∈ Q ^ k * W.part j := by
    obtain ⟨D, hD⟩ := UpperTheory.V.Alg.part_filt_split W.wt j
    intro k
    induction k with
    | zero =>
      refine ⟨0, fun n hn _ => ?_⟩
      rw [pow_zero, one_mul]
      exact ⟨_, UpperTheory.V.Alg.part_filt_monomial W.wt hn 1, rfl⟩
    | succ k ih =>
      obtain ⟨N, hN⟩ := ih
      refine ⟨N + D + 1, fun n hn hd => ?_⟩
      obtain ⟨g, n', hg0, hgw, hgd, rfl⟩ := hD n hn (by omega)
      rw [map_add] at hd
      rw [map_add, hgw, zero_add] at hn
      have h1 := hN n' hn (by omega)
      have h2 : W.π (monomial g 1) ∈ Q := by
        refine ⟨⟨_, UpperTheory.V.Alg.part_filt_monomial W.wt hgw 1, rfl⟩, ?_⟩
        intro hu
        have h3 : IsUnit (monomial g (1 : O)) := hloc.map_nonunit _ hu
        rw [MvPowerSeries.isUnit_iff_constantCoeff, ← coeff_zero_eq_constantCoeff_apply, coeff_monomial,
          if_neg hg0.symm] at h3
        exact not_isUnit_zero h3
      have h4 : (monomial (g + n') (1 : O) : MvPowerSeries W.σ O) = monomial g 1 * monomial n' 1 := by
        rw [monomial_mul_monomial, one_mul]
      rw [h4, map_mul, pow_succ', mul_assoc]
      exact Submodule.mul_mem_mul h2 h1

  have hconst : ∀ k : ℕ, ∀ c ∈ maximalIdeal O ^ k, algebraMap O R c ∈ Q ^ k := by
    intro k
    induction k with
    | zero =>
      intro c _
      rw [pow_zero]
      exact Submodule.algebraMap_mem c
    | succ k ih =>
      intro c hc
      rw [pow_succ] at hc
      refine Submodule.mul_induction_on hc (fun a ha b hb => ?_) (fun a b ha hb => ?_)
      · rw [map_mul, pow_succ]
        refine Submodule.mul_mem_mul (ih a ha) ⟨?_, ?_⟩
        · rw [Algebra.algebraMap_eq_smul_one]
          exact Submodule.smul_mem _ _ hone
        · exact fun hu => hb (IsLocalHom.map_nonunit (f := algebraMap O R) b hu)
      · rw [map_add]
        exact add_mem ha hb
  obtain ⟨N, hN⟩ := hmono k
  refine ⟨N + k, ?_⟩
  rintro x ⟨⟨f, hf, hfx⟩, hx⟩
  have hfx' : W.π f = x := hfx
  have hle : maximalIdeal R ≤ (maximalIdeal (MvPowerSeries W.σ O)).map W.π := by
    intro y hy
    obtain ⟨g, rfl⟩ := W.surj y
    exact Ideal.mem_map_of_mem _ fun hu => hy (hu.map W.π)
  have hx' : x ∈ (maximalIdeal (MvPowerSeries W.σ O) ^ (N + k)).map W.π := by
    rw [Ideal.map_pow]
    exact Ideal.pow_right_mono hle _ hx
  obtain ⟨g, hg, hgx⟩ := (Ideal.mem_map_iff_of_surjective W.π W.surj).1 hx'
  set h := wcomp O W.wt j g with hh
  have hhx : W.π h = x := by
    have h1 : f - g ∈ RingHom.ker W.π := by
      rw [RingHom.mem_ker, map_sub, hfx', hgx, sub_self]
    have h2 := W.homog _ h1 j
    have h3 : wcomp O W.wt j f = f := by
      ext n
      rw [coeff_wcomp]
      split_ifs with hn
      · rfl
      · exact (hf n hn).symm
    rw [map_sub, h3, RingHom.mem_ker, map_sub, hfx', sub_eq_zero] at h2
    exact h2.symm
  have hhj : h ∈ wtPart O W.wt j := wcomp_mem O W.wt j g
  have hhc : ∀ n : W.σ →₀ ℕ, n.degree < N → coeff n h ∈ maximalIdeal O ^ k := by
    intro n hn
    rw [hh, coeff_wcomp]
    split_ifs
    · exact Ideal.pow_le_pow_right (by omega) (UpperTheory.V.Alg.part_filt_coeff _ g hg n)
    · exact zero_mem _
  have hfin : {n : W.σ →₀ ℕ | n.degree < N}.Finite :=
    (Finsupp.finite_of_degree_le N).subset fun n (hn : n.degree < N) => hn.le
  set low : MvPowerSeries W.σ O := ∑ n ∈ hfin.toFinset, monomial n (coeff n h) with hlow
  have hlowc : ∀ m, coeff m low = if m.degree < N then coeff m h else 0 := by
    intro m
    rw [hlow, map_sum]
    simp only [coeff_monomial]
    rw [Finset.sum_ite_eq]
    simp
  obtain ⟨μ, hμ, hspan⟩ := UpperTheory.V.Alg.part_span O R X W
    {n | Finsupp.weight W.wt n = j ∧ N ≤ n.degree} j fun n hn => hn.1
  obtain ⟨b, hb, hhigh⟩ := hspan (h - low) (by
    intro n hn
    rw [map_sub, hlowc]
    split_ifs with hd
    · exact sub_self _
    · rw [sub_zero]
      exact hhj n fun hw => hn ⟨hw, not_lt.1 hd⟩)
  have hsum : x = W.π low + W.π (h - low) := by rw [← map_add, add_sub_cancel, hhx]
  rw [hsum]
  refine add_mem ?_ ?_
  · rw [hlow, map_sum]
    refine Submodule.sum_mem _ fun n hn => ?_
    by_cases hw : Finsupp.weight W.wt n = j
    · have h1 : W.π (monomial n (coeff n h)) = algebraMap O R (coeff n h) * 1 * W.π (monomial n 1) := by
        rw [mul_one, ← Algebra.smul_def, ← map_smul, ← map_smul, smul_eq_mul, mul_one]
      rw [h1]
      exact Submodule.mul_mem_mul
        (Submodule.mul_mem_mul (hconst k _ (hhc n (hfin.mem_toFinset.1 hn))) hone)
        ⟨_, UpperTheory.V.Alg.part_filt_monomial W.wt hw 1, rfl⟩
    · rw [hhj n hw, map_zero, map_zero]
      exact zero_mem _
  · rw [hhigh, map_sum]
    refine Submodule.sum_mem _ fun m hm => ?_
    rw [map_mul, mul_comm (Q ^ k), mul_assoc]
    exact Submodule.mul_mem_mul ⟨_, hb m, rfl⟩ (hN m (hμ hm).1 (hμ hm).2)

private theorem rep_ex {l : ℕ} [Fact l.Prime] {k : Type} [Field k] [CharP k l] {F : Frame l k} {C₀ : ℕ}

      ∀ g, ρ.r g 1 0 = a g := by

  obtain ⟨hlc, hmul, hstar, hloc⟩ := ha

    simpa using h1

  have hpstar : ∀ g, p (F.star g) = q g := fun g => by rw [hp, hq, hΨstar]
  have hqstar : ∀ g, q (F.star g) = p g := fun g => by rw [hp, hq, hΨstar, inv_inv]

      ψ := fun v => ![χ v, (χ v)⁻¹]
      pol := by
        intro g
        rw [hr, hr, hpstar, hqstar, hstar]
        exact pol2 _ _ _
      diag := by
        intro i j hij
        rw [hr, hr]
        fin_cases i <;> fin_cases j
        · exact absurd rfl hij
        · simp
        · simp [hstar]
        · exact absurd rfl hij
      ord := by
        intro v g
        obtain ⟨γ, hγ⟩ := hloc v
        refine ord2 _ _ γ (fun s => ?_) g
        rw [hr, hγ]
        simp [χ, hp, hq, Algebra.smul_def, Frame.loc] }
  refine ⟨ρ, ⟨?_, ?_⟩, ?_, ?_, ?_⟩
  · intro g i
    show r g i i = _
    rw [hr]
    fin_cases i <;> simp [w, hp, hq]
  · intro v i g
    fin_cases i <;> simp [ρ, χ, w]
  · apply Ideal.span_eq_bot.mpr
    rintro _ ⟨g, i, j, hij, rfl⟩
    show r g i j = 0
    rw [hr]
    fin_cases i <;> fin_cases j <;> simp at hij ⊢
  · have hl : IsLocallyConstant fun g : F.G0 => F.Ψ (g : F.Γ) := isLocallyConstant_psi_restrict_of_isOpen_ker hF.Ψ_open
    intro i j
    show IsLocallyConstant fun g => r g i j

    · exact IsLocallyConstant.const _
    · exact hlc
    · exact hl.comp fun z => algebraMap k Q ((z⁻¹ : kˣ) : k)

    show r g 1 0 = a g
    rw [hr]
    simp

private theorem cont_sub {Γ S C : Type} [TopologicalSpace Γ] [CommRing S] [IsNoetherianRing S] [CommRing C]
    [Algebra S C] [Module.Finite S C] (I : Ideal S) (J : Ideal C)
    (h : ∀ N, ∀ c ∈ J ^ N, c ∈ (I ^ N • ⊤ : Submodule S C)) (a : Γ → C) (ha : AdicCont J a) (N : ℕ) :
    IsLocallyConstant fun g => (I ^ N • Submodule.span S (Set.range a)).mkQ (a g) := by
  classical
  obtain ⟨k, hk⟩ := Ideal.exists_pow_inf_eq_pow_smul I (Submodule.span S (Set.range a))
  set M := Submodule.span S (Set.range a)
  let φ : Γ → C ⧸ J ^ (N + k) := fun g => Ideal.Quotient.mk (J ^ (N + k)) (a g)
  let f : Γ → C ⧸ (I ^ N • M) := fun g => (I ^ N • M).mkQ (a g)
  have key : ∀ g g', φ g = φ g' → f g = f g' := by
    intro g g' hgg
    have h1 : a g - a g' ∈ J ^ (N + k) := Ideal.Quotient.eq.1 hgg
    have h2 : a g - a g' ∈ (I ^ (N + k) • ⊤ : Submodule S C) ⊓ M :=
      ⟨h _ _ h1, M.sub_mem (Submodule.subset_span ⟨g, rfl⟩) (Submodule.subset_span ⟨g', rfl⟩)⟩
    rw [hk (N + k) (by omega), Nat.add_sub_cancel] at h2
    have h3 : a g - a g' ∈ I ^ N • M := Submodule.smul_mono le_rfl inf_le_right h2
    exact (Submodule.Quotient.eq _).2 h3
  have hf : f = (fun x => if hx : ∃ g, φ g = x then f hx.choose else 0) ∘ φ := by
    funext g
    have hx : ∃ g', φ g' = φ g := ⟨g, rfl⟩
    simp only [Function.comp_apply, dif_pos hx]
    exact key _ _ hx.choose_spec.symm
  change IsLocallyConstant f
  rw [hf]
  exact (ha (N + k)).comp _

private theorem two_zero {K : Type} [Field K] {x y s t : K} (h : s ≠ t) (h1 : x + y = 0) (h2 : s * x + t * y = 0) :
    x = 0 ∧ y = 0 := by
  have hy : y = -x := by linear_combination h1
  subst hy
  have hx : (s - t) * x = 0 := by linear_combination h2
  rcases mul_eq_zero.1 hx with h3 | h3
  · exact absurd (sub_eq_zero.1 h3) h
  · exact ⟨h3, by simp [h3]⟩

private theorem or_zero {G K : Type} [Field K] {a b : G → K} (h : ∀ g g', a g * b g' = 0) :
    (∀ g, a g = 0) ∨ ∀ g, b g = 0 := by
  by_contra hc
  push_neg at hc
  obtain ⟨⟨g, hg⟩, ⟨g', hg'⟩⟩ := hc
  exact mul_ne_zero hg hg' (h g g')

private theorem tri_of_bar {l : ℕ} [Fact l.Prime] {k : Type} [Field k] [CharP k l] {F : Frame l k} {C₀ n : ℕ}

      simp [← hpdef]
    have hq : f ↑p - f ↑p⁻¹ ≠ 0 := by

      apply hp
      have e2 : f ((p : k) * p) = f ((p : k) * ↑p⁻¹) := by
        rw [map_mul, map_mul, sub_eq_zero.1 e]

    have hA : ∀ g h : F.G0, ρ.r g 0 1 * ρ.r h 1 0 = 0 ∧ ρ.r g 0 2 * ρ.r h 2 0 = 0 := by

      rw [← hst.1 g, s10]; exact h _

      intro a b c g i j hij
      fin_cases i <;> fin_cases j <;> simp at hij
      · exact a g
      · exact b g
      · exact c g
    rcases or_zero (fun g h => (hA g h).1) with a01 | a10
    · have a12 := t12 a01
      rcases or_zero (fun g h => (hA g h).2) with a02 | a20

private theorem wsum {G : Type} {n : ℕ} (hn : n = 2 ∨ n = 3) (L : List (G × Fin n × Fin n))
    (h : ∀ s ∈ L, s.2.1 ≤ s.2.2) (h0 : (L.map fun s => w n s.2.1 - w n s.2.2).sum = 0) : ∀ s ∈ L, s.2.1 = s.2.2 := by
  have hw : ∀ i j : Fin n, i ≤ j → 0 ≤ w n i - w n j ∧ (w n i - w n j = 0 → i = j) := by
    intro i j hij
    have hij' : (i : ℕ) ≤ j := hij
    rcases hn with rfl | rfl
    · simp only [w]
      refine ⟨by norm_num; omega, fun e => Fin.ext ?_⟩
      norm_num at e
      omega
    · simp only [w]
      refine ⟨by norm_num; omega, fun e => Fin.ext ?_⟩
      norm_num at e
      omega
  induction L with
  | nil => simp
  | cons s L ih =>
    intro s' hs'
    have hs := hw s.2.1 s.2.2 (h s (List.mem_cons_self ..))
    have hL : ∀ t ∈ L, t.2.1 ≤ t.2.2 := fun t ht => h t (List.mem_cons_of_mem _ ht)
    have hnn : 0 ≤ (L.map fun s => w n s.2.1 - w n s.2.2).sum := by
      apply List.sum_nonneg
      intro x hx
      obtain ⟨t, ht, rfl⟩ := List.mem_map.1 hx
      exact (hw _ _ (hL t ht)).1
    rw [List.map_cons, List.sum_cons] at h0
    have e1 : w n s.2.1 - w n s.2.2 = 0 := by linarith [hs.1]
    have e2 : (L.map fun s => w n s.2.1 - w n s.2.2).sum = 0 := by linarith [hs.1]
    rcases List.mem_cons.1 hs' with rfl | hs'
    · exact hs.2 e1
    · exact ih hL e2 s' hs'

private theorem cyc_mem {l : ℕ} [Fact l.Prime] {k : Type} [Field k] [CharP k l] {F : Frame l k} {C₀ n : ℕ}
    (hF : F.Ok C₀) (hn : n = 2 ∨ n = 3) {C : Type} [CommRing C] [IsLocalRing C] (f : k →+* ResidueField C)
    (ρ : NRep F n C) (hρ : (ρ.map (residue C)).IsBar f) (L : List (F.G0 × Fin n × Fin n))
    (h0 : (L.map fun s => w n s.2.1 - w n s.2.2).sum = 0) (h1 : ∃ s ∈ L, s.2.1 ≠ s.2.2) :
    (L.map fun s => ρ.r s.1 s.2.1 s.2.2).prod ∈ maximalIdeal C := by
  have hr : ∀ g i j, (ρ.map (residue C)).r g i j = residue C (ρ.r g i j) := fun _ _ _ => rfl
  rw [← IsLocalRing.residue_eq_zero_iff, map_list_prod, List.map_map]
  apply List.prod_eq_zero
  rcases tri_of_bar hF hn f (ρ.map (residue C)) hρ with ht | ht
  · by_cases hex : ∃ s ∈ L, s.2.2 < s.2.1
    · obtain ⟨s, hs, hlt⟩ := hex
      refine List.mem_map.2 ⟨s, hs, ?_⟩
      simp only [Function.comp_apply]
      rw [← hr]
      exact ht _ _ _ hlt
    · push_neg at hex
      exfalso
      obtain ⟨s, hs, hne⟩ := h1
      exact hne (wsum hn L hex h0 s hs)
  · by_cases hex : ∃ s ∈ L, s.2.1 < s.2.2
    · obtain ⟨s, hs, hlt⟩ := hex
      refine List.mem_map.2 ⟨s, hs, ?_⟩
      simp only [Function.comp_apply]
      rw [← hr]
      exact ht _ _ _ hlt
    · push_neg at hex
      exfalso
      obtain ⟨s, hs, hne⟩ := h1
      have hL' : ∀ t ∈ L.map (fun s => (s.1, s.2.2, s.2.1)), t.2.1 ≤ t.2.2 := by
        intro t ht
        obtain ⟨s, hs, rfl⟩ := List.mem_map.1 ht
        exact hex s hs
      have h0' : ((L.map (fun s => (s.1, s.2.2, s.2.1))).map fun s => w n s.2.1 - w n s.2.2).sum = 0 := by
        have e : ((L.map (fun s => (s.1, s.2.2, s.2.1))).map fun s => w n s.2.1 - w n s.2.2) =
            (L.map fun s => w n s.2.1 - w n s.2.2).map Neg.neg := by
          simp [List.map_map, Function.comp_def]
        have hneg : ∀ m : List ℤ, (m.map Neg.neg).sum = -m.sum := by
          intro m
          induction m with
          | nil => simp
          | cons a m ih => simp only [List.map_cons, List.sum_cons, ih]; ring
        rw [e, hneg, h0, neg_zero]
      exact hne (wsum hn _ hL' h0' (s.1, s.2.2, s.2.1) (List.mem_map.2 ⟨s, hs, rfl⟩)).symm

private theorem lc_lemma {X A B : Type} [TopologicalSpace X] (φ : X → A) (g : X → B)

  exact hφ.comp _
