structure Gen (η : G ℚ Ω →* ℤˣ) (Ψ : G ℚ Ω →* Oˣ) (σ : G ℚ Ω) (B : ValuationSubring Ω) (M : ℕ) (Hb : Subgroup (G ℚ Ω)) : Prop where
  odd : l ≠ 2
  fin : Finite (ResidueField O)
  M_ne : M ≠ 0
  l_dvd : l ∣ M
  two_dvd : 2 ∣ M
  cc : IsCC σ
  over_l : B.valuation (l : Ω) < 1
  quad : ∃ (D : ℕ) (δ : Ω), 0 < D ∧ (∀ q : ℕ, q.Prime → q ∣ D → q ∣ M) ∧ δ ^ 2 = -(D : Ω) ∧ ∀ s : G ℚ Ω, s δ = (η s : ℤ) • δ

  even : barChar O Ψ σ = 1

  g4 : ¬ (omO Ω l O).ker ⊓ (barChar O Ψ).ker ≤ η.ker
  g5 : ∃ s ∈ Dec (F := ℚ) B, η s = 1 ∧ barChar O Ψ s ^ 2 ≠ 1

  Hb_ab : commutator (G ℚ Ω) ≤ Hb
  Hb_idx : ∃ a : ℕ, Hb.index = l ^ a

def Hth (w : ℕ) : Subgroup (G ℚ Ω) where
  carrier := {s | ∃ a : ℕ, ∀ ζ : Ω, ζ ^ w = 1 → s ζ = ζ ^ a ^ l} ∩ {s | ∃ a : ℕ, ∀ ζ : Ω, ζ ^ w = 1 → s⁻¹ ζ = ζ ^ a ^ l}
  one_mem' := ⟨⟨1, by simp⟩, ⟨1, by simp⟩⟩
  mul_mem' := by
    rintro s t ⟨⟨a, ha⟩, ⟨a', ha'⟩⟩ ⟨⟨b, hb⟩, ⟨b', hb'⟩⟩
    refine ⟨⟨a * b, fun ζ h => ?_⟩, ⟨b' * a', fun ζ h => ?_⟩⟩
    · rw [AlgEquiv.mul_apply, hb ζ h, map_pow, ha ζ h, ← pow_mul, mul_pow]
    · rw [mul_inv_rev, AlgEquiv.mul_apply, ha' ζ h, map_pow, hb' ζ h, ← pow_mul, mul_pow]
  inv_mem' := by
    rintro s ⟨h1, h2⟩
    exact ⟨h2, by show ∃ a : ℕ, ∀ ζ : Ω, ζ ^ w = 1 → s⁻¹⁻¹ ζ = ζ ^ a ^ l; rw [inv_inv]; exact h1⟩

def Hlay (m : ℕ) : Subgroup (G ℚ Ω) := (cycMod ℚ Ω l (m + 1) ^ (l - 1)).ker

structure FrobeniusSplitPrimeDatum (η : G ℚ Ω →* ℤˣ) (M : ℕ) (H₀ : Subgroup (G ℚ Ω)) : Prop where
  isOpen : IsOpen (H₀ : Set (G ℚ Ω))
  idx : H₀.index = 2
  real : ∀ τ : G ℚ Ω, IsCC τ → τ ∈ H₀
  loc : ∀ B' : ValuationSubring Ω, B'.valuation (M : Ω) < 1 → Dec (F := ℚ) B' ⊓ H₀ = Dec (F := ℚ) B' ⊓ η.ker
  ram : ∃ B' : ValuationSubring Ω, ¬ B'.valuation (M : Ω) < 1 ∧ ¬ Ine (F := ℚ) B' ≤ H₀

structure FrobeniusWitnessPrime (η : G ℚ Ω →* ℤˣ) (Ψ : G ℚ Ω →* Oˣ) (M w : ℕ) : Prop where
  prime : w.Prime
  cop : ¬ w ∣ M
  one : w ≡ 1 [MOD l]
  split : ∀ B' : ValuationSubring Ω, B'.valuation (w : Ω) < 1 →
    Dec (F := ℚ) B' ≤ η.ker ⊓ (barChar O Ψ).ker ⊓ (omO Ω l O).ker

def Upl (Hb H₀ : Subgroup (G ℚ Ω)) (w m : ℕ) : Subgroup (G ℚ Ω) := H₀ ⊓ Hth Ω l w ⊓ Hb ⊓ Hlay Ω l m

      ∀ v s, galEmb ℚ (W.Lp L) Ω (ι v s) = k.j v s

def upl (L : Type) [Field L] [NumberField L] [Algebra L Ω] : Subgroup (G ℚ Ω) where
  carrier := {s | ∀ x ∈ maximalRealSubfield L, s (algebraMap L Ω x) = algebraMap L Ω x}
  one_mem' _ _ := rfl
  mul_mem' {a b} ha hb x hx := by rw [AlgEquiv.mul_apply, hb x hx, ha x hx]
  inv_mem' {a} ha x hx := a.symm_apply_eq.2 (ha x hx).symm

structure GalLink (η : G ℚ Ω →* ℤˣ) (Ψ : G ℚ Ω →* Oˣ) (σ g₁ : G ℚ Ω) (B : ValuationSubring Ω) (M : ℕ)
    (L : Type) [Field L] [NumberField L] [Algebra L Ω] (Φ : Fr.Frame l (ResidueField O)) where
  ab : IsAbelianGalois ℚ L
  gal : galOf ℚ L Ω = upl Ω L ⊓ η.ker
  unr : ∀ B' : ValuationSubring Ω, Ine (F := ℚ) B' ⊓ upl Ω L ≤ η.ker
  split : ∀ B' : ValuationSubring Ω, B'.valuation (M : Ω) < 1 → Dec (F := ℚ) B' ⊓ upl Ω L ≤ η.ker
  teich : ∀ B' : ValuationSubring Ω, ¬ B'.valuation (l : Ω) < 1 → ∀ s ∈ Ine (F := ℚ) B' ⊓ upl Ω L,
    barChar O Ψ s = 1 → Ψ s = 1
  p : upl Ω L →* Φ.Γ
  p_cont : Continuous p
  p_surj : Function.Surjective p
  p_ker : p.ker = (Subgroup.normalClosure {s : upl Ω L | ∃ B' : ValuationSubring Ω, ¬ B'.valuation (l : Ω) < 1 ∧
    s.1 ∈ Ine (F := ℚ) B' ∧ η s = 1 ∧ barChar O Ψ s = 1}).topologicalClosure
  del : ∀ s, Φ.del (p s) = η s
  eps : ∀ s, Φ.eps (p s) = cycZ ℚ Ω l s
  psi : ∀ s, Φ.Ψ (p s) = barChar O Ψ s
  c : ∃ h : σ ∈ upl Ω L, p ⟨σ, h⟩ = Φ.c
  g0 : ∃ h : g₁ ∈ upl Ω L, p ⟨g₁, h⟩ = Φ.g0
  pl : Φ.V → ValuationSubring Ω
  pl_conj : ∀ v, ∃ γ : G ℚ Ω, η γ = 1 ∧ pl v = γ • B
  pl_all : ∀ B' : ValuationSubring Ω, B'.valuation (l : Ω) < 1 → ∃! v, ∃ u ∈ upl Ω L, B' = u • pl v
  j : ∀ v, Φ.Gv v →* upl Ω L
  j_cont : ∀ v, Continuous (j v)
  j_inj : ∀ v, Function.Injective (j v)
  j_range : ∀ v (s : upl Ω L), s ∈ (j v).range ↔ s.1 ∈ Dec (F := ℚ) (pl v)
  j_loc : ∀ v g, Φ.loc v g = p (j v g)
  j_ine : ∀ v g, g ∈ Φ.Iv v ↔ (j v g).1 ∈ Ine (F := ℚ) (pl v)
  dv : ∀ v, Φ.dv v = (upl Ω L).relIndex (Dec (F := ℚ) (pl v))
  mu : ∀ v, ∃ g, cycMod ℚ Ω l 1 (j v g) ≠ 1

def GalLink.p0 {η : G ℚ Ω →* ℤˣ} {Ψ : G ℚ Ω →* Oˣ} {σ g₁ : G ℚ Ω} {B : ValuationSubring Ω} {M : ℕ}
    {L : Type} [Field L] [NumberField L] [Algebra L Ω] {Φ : Fr.Frame l (ResidueField O)} (k : GalLink η Ψ σ g₁ B M L Φ)
    (s : G L Ω) : Φ.G0 :=

    Sd.FieldPairCharacterTripleConditions K L σ N w χ' η' θ j ∧ χ' ≠ η' ∧ ¬ l ∣ orderOf χ' ∧ (∀ s, (η' s).val = (((η s).val : ℤ) : Qbar l)) ∧
    (∀ s, ‖((cycBar ℚ Ω l ^ j * χ') s).val - (Ψ s).val.1‖ < 1) ∧ ¬ (l : ℤ) ∣ discr K

def HasFrame (c : Coef l) [CharP (ResidueField c.O) l] (η : G ℚ Ω →* ℤˣ) (Ψ : G ℚ Ω →* c.Oˣ) (σ : G ℚ Ω) (B : ValuationSubring Ω)

def HostFrameEx : Prop :=
  ∀ (Ω : Type) [Field Ω] [CharZero Ω] [IsAlgClosure ℚ Ω] (l : ℕ) [Fact l.Prime] (c : Coef l) [CharP (ResidueField c.O) l]

def symSquareMatrixHom : Matrix (Fin 2) (Fin 2) R →* Matrix (Fin 3) (Fin 3) R where
  toFun g := !![g 0 0 ^ 2, g 0 0 * g 0 1, g 0 1 ^ 2;
    2 * g 0 0 * g 1 0, g 0 0 * g 1 1 + g 0 1 * g 1 0, 2 * g 0 1 * g 1 1;
    g 1 0 ^ 2, g 1 0 * g 1 1, g 1 1 ^ 2]
  map_one' := by ext i j; fin_cases i <;> fin_cases j <;> simp
  map_mul' g h := by
    ext i j; fin_cases i <;> fin_cases j <;> simp [Matrix.mul_apply, Fin.sum_univ_succ] <;> ring

private theorem gh_two {O : Type} [CommRing O] [IsDomain O] {N : ℕ} [NeZero N] (ψ : DirichletCharacter O N) (hc : 4 ≤ c2 ψ) :
    ∃ (h : 4 ∣ N) (u : (ZMod N)ˣ), ZMod.unitsMap h u = -1 ∧ (psi2 ψ).toUnitHom u = 1 := by
  obtain ⟨-, -, -, h1, h2, -⟩ := compCond O N ψ 2 Nat.prime_two
  have h4 : 4 ∣ N := (pow_dvd_pow 2 (show 2 ≤ c2 ψ by omega)).trans h2
  obtain ⟨k, hk⟩ := (compOrd O N ψ 2 Nat.prime_two).2.1
  have hm : ZMod.unitsMap h4 (-1) = -1 := by ext; simp [ZMod.unitsMap_def]
  refine ⟨h4, ?_⟩
  have hv : ∃ v : (ZMod N)ˣ, ZMod.unitsMap h4 v = 1 ∧ (psi2 ψ).toUnitHom v ≠ 1 := by
    by_contra hn
    push Not at hn
    have h5 : (psi2 ψ).conductor ∣ 4 := by
      rw [← DirichletCharacter.mem_conductorSet_iff_conductor_dvd _ h4, DirichletCharacter.mem_conductorSet_iff,
        DirichletCharacter.factorsThrough_iff_ker_unitsMap h4]
      exact hn
    rw [← h1] at h5
    have := (Nat.pow_dvd_pow_iff_le_right one_lt_two).1 (show 2 ^ c2 ψ ∣ 2 ^ 2 from h5)
    omega
  obtain ⟨v, hv1, hv2⟩ := hv
  have hv3 : (psi2 ψ).toUnitHom v ^ 2 ^ k = 1 := by
    ext
    rw [Units.val_pow_eq_pow_val, MulChar.coe_toUnitHom, ← MulChar.pow_apply_coe, ← hk, psi2, pow_orderOf_eq_one,
      MulChar.one_apply_coe, Units.val_one]
  obtain ⟨j, -, hj⟩ := (Nat.dvd_prime_pow Nat.prime_two).1 (orderOf_dvd_of_pow_eq_one hv3)
  obtain ⟨i, rfl⟩ : ∃ i, j = i + 1 := Nat.exists_eq_succ_of_ne_zero (by
    rintro rfl
    exact hv2 (orderOf_eq_one_iff.1 hj))
  have h6 : (psi2 ψ).toUnitHom (v ^ 2 ^ i) = -1 := by
    have a1 : (psi2 ψ).toUnitHom (v ^ 2 ^ i) ^ 2 = 1 := by rw [map_pow, ← pow_mul, ← pow_succ, ← hj, pow_orderOf_eq_one]
    have a2 : (psi2 ψ).toUnitHom (v ^ 2 ^ i) ≠ 1 := by
      rw [map_pow]
      refine pow_ne_one_of_lt_orderOf (by positivity) ?_
      rw [hj]
      exact Nat.pow_lt_pow_right one_lt_two (Nat.lt_succ_self i)
    exact (gh_sq a1).resolve_left a2
  have h7 : (psi2 ψ).toUnitHom (-1) ^ 2 = 1 := by rw [← map_pow]; simp
  rcases gh_sq h7 with h | h
  · exact ⟨-1, hm, h⟩
  · exact ⟨-1 * v ^ 2 ^ i, by rw [map_mul, map_pow, hv1, one_pow, mul_one, hm], by rw [map_mul, h, h6]; simp⟩

private theorem kernel_not_contained_eta (l : ℕ) [hl : Fact l.Prime] (hl2 : l ≠ 2) (c : Coef l) [CharP (ResidueField c.O) l] {N : ℕ} [NeZero N] (e : ℕ)

    exact hu2
  have h3 : UnitaryThreeInertia.eta4 Ω s = 1 := hle ⟨h1, by
    show barChar c.O _ s = 1
    rw [gh_bar, h1, one_pow, one_mul, barChar, MonoidHom.comp_apply, h2, map_one]⟩

structure LLink {V : Type} {Gv : V → Type} [∀ v, Group (Gv v)] [∀ v, TopologicalSpace (Gv v)] (ι : ∀ v, Gv v →* G F Ω)
    (A : V → ValuationSubring Ω) (pl : V → HeightOneSpectrum (𝓞 F)) : Prop where
  isOver : ∀ v, IsPlaceOver (A v) (pl v)
  inj : Function.Injective pl
  mem : ∀ w, w ∈ Set.range pl ↔ ((l : ℕ) : 𝓞 F) ∈ w.asIdeal
  cont : ∀ v, Continuous (ι v)
  ι_inj : ∀ v, Function.Injective (ι v)
  range : ∀ v, (ι v).range = Dec (F := F) (A v)

structure FLink {k : Type} [Field k] (Φ : Fr.Frame l k) (S : Finset (HeightOneSpectrum (𝓞 F))) (K : Subgroup (G F Ω))
    (π : G F Ω →* Φ.Γ) (ι : ∀ v, Φ.Gv v →* G F Ω) (A : Φ.V → ValuationSubring Ω) (pl : Φ.V → HeightOneSpectrum (𝓞 F))
    (e : Φ.Inf ≃ InfinitePlace F) (c : Φ.Inf → G F Ω) : Prop where
  real : IsTotallyReal F
  g : GLink l S K π
  only : ∀ w ∈ S, ((l : ℕ) : 𝓞 F) ∈ w.asIdeal
  loc : LLink l ι A pl
  loc_eq : ∀ v s, Φ.loc v s = π (ι v s)
  ine : ∀ v, Φ.Iv v = (Ine (F := F) (A v)).comap (ι v)
  dv : ∀ v, Φ.dv v = locDeg (pl v)
  eps : ∀ g, Φ.eps (π g) = cycZ F Ω l g
  cc : ∀ w, IsCCAt (e w) (c w)
  cw : ∀ w, Φ.cw w = π (c w)
  del_open : IsOpen (Φ.del.ker : Set Φ.Γ)
  del_c : Φ.del Φ.c = -1
  K_del : ∀ g ∈ K, Φ.del (π g) = 1
  K_Ψ : ∀ g ∈ K, Φ.Ψ (π g) = 1

theorem gLink_quot (hl : l ≠ 2) {S : Finset (HeightOneSpectrum (𝓞 F))} (hS : ContainsAbove S l) {K : Subgroup (G F Ω)}
    (h1 : K.Normal) (h2 : IsOpen (K : Set (G F Ω))) (h3 : ¬ l ∣ K.index)
    [(tameN (S : Set (HeightOneSpectrum (𝓞 F))) K).Normal] :
    GLink l S K (QuotientGroup.mk' (tameN (S : Set (HeightOneSpectrum (𝓞 F))) K)) := by
  refine ⟨hl, hS, h1, h2, h3, ?_, QuotientGroup.mk'_surjective _, QuotientGroup.ker_mk' _⟩
  exact continuous_quot_mk
