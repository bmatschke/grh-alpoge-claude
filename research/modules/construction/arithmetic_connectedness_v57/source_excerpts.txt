class IsCO (O R : Type) [CommRing O] [IsLocalRing O] [CommRing R] [Algebra O R] : Prop
    extends IsLocalRing R, IsNoetherianRing R where
  complete : IsAdicComplete (maximalIdeal R) R
  isLocalHom : IsLocalHom (algebraMap O R)
  res_surj : ∀ x : R, ∃ a : O, x - algebraMap O R a ∈ maximalIdeal R

attribute [instance] IsCO.complete IsCO.isLocalHom

instance IsCO.self (O : Type) [CommRing O] [IsLocalRing O] [IsNoetherianRing O]
    [IsAdicComplete (maximalIdeal O) O] : IsCO O O where
  complete := inferInstance
  isLocalHom := ⟨fun _ h => h⟩
  res_surj x := ⟨x, by simp⟩

variable (O R : Type) [CommRing O] [IsLocalRing O] [CommRing R] [Algebra O R] [IsCO O R]

theorem resMap_bijective : Function.Bijective (ResidueField.map (algebraMap O R)) := by
  refine ⟨(ResidueField.map (algebraMap O R)).injective, fun y => ?_⟩
  obtain ⟨x, rfl⟩ := residue_surjective y
  obtain ⟨a, ha⟩ := IsCO.res_surj (O := O) x
  refine ⟨residue O a, ?_⟩
  rw [ResidueField.map_residue]
  exact ((Ideal.Quotient.eq).2 ha).symm

def res : R →ₐ[O] ResidueField O :=
  { ((RingEquiv.ofBijective _ (resMap_bijective O R)).symm.toRingHom.comp (residue R)) with
    commutes' := fun a => by
      show (RingEquiv.ofBijective _ (resMap_bijective O R)).symm (residue R (algebraMap O R a)) = residue O a
      rw [RingEquiv.symm_apply_eq]
      exact (ResidueField.map_residue _ _).symm }

theorem res_eq_zero_iff (x : R) : res O R x = 0 ↔ x ∈ maximalIdeal R := by
  show (RingEquiv.ofBijective _ (resMap_bijective O R)).symm (residue R x) = 0 ↔ _
  rw [map_eq_zero_iff _ (RingEquiv.injective _), residue_eq_zero_iff]

theorem res_map {R' : Type} [CommRing R'] [Algebra O R'] [IsCO O R'] (f : R →ₐ[O] R') (x : R) :
    res O R' (f x) = res O R x := by
  obtain ⟨a, ha⟩ := IsCO.res_surj (O := O) x
  have h1 : res O R x = algebraMap O _ a := by
    have := (res_eq_zero_iff O R _).2 ha
    rwa [map_sub, AlgHom.commutes, sub_eq_zero] at this
  have hm : ∀ y ∈ maximalIdeal R, f y ∈ maximalIdeal R' := fun y hy => by
    obtain ⟨b, hb⟩ := IsCO.res_surj (O := O) (f y)
    by_cases hu : IsUnit b
    · exfalso
      have h2 : IsUnit (y - algebraMap O R b) := by
        by_contra hn
        have h3 : y - algebraMap O R b ∈ maximalIdeal R := hn
        have h4 : algebraMap O R b ∈ maximalIdeal R := by simpa using Ideal.sub_mem _ hy h3
        exact h4 (hu.map _)
      have h5 := h2.map f
      rw [map_sub, AlgHom.commutes] at h5
      exact hb h5
    · have h6 : algebraMap O R' b ∈ maximalIdeal R' := fun h => hu (IsLocalHom.map_nonunit b h)
      simpa using Ideal.add_mem _ hb h6
  have h2 := (res_eq_zero_iff O R' _).2 (hm _ ha)
  rw [map_sub, map_sub, AlgHom.commutes, AlgHom.commutes, sub_eq_zero] at h2
  rw [h2, h1]

end Res

def ordProd {H K : Type} [CommRing K] {n : ℕ} (r : H → Matrix (Fin n) (Fin n) K) (ψ : Fin n → H → K) (g : Fin n → H) :
    Matrix (Fin n) (Fin n) K := (List.ofFn fun i => r (g i) - ψ i (g i) • (1 : Matrix (Fin n) (Fin n) K)).prod

theorem ordProd_map {H K K' : Type} [CommRing K] [CommRing K'] {n : ℕ} (f : K →+* K')
    (r : H → Matrix (Fin n) (Fin n) K) (ψ : Fin n → H → K) (g : Fin n → H) :
    ordProd (fun s => f.mapMatrix (r s)) (fun i s => f (ψ i s)) g = f.mapMatrix (ordProd r ψ g) := by
  simp only [ordProd, map_list_prod, List.map_ofFn]
  congr 1
  refine congrArg List.ofFn (funext fun i => ?_)
  simp only [Function.comp_apply, map_sub]
  congr 1
  ext a b
  by_cases h : a = b <;> simp [h, Matrix.one_apply]

@[ext] structure NRep (K : Type) [CommRing K] where
  r : F.G0 →* Matrix (Fin n) (Fin n) K
  ψ : ∀ v, Fin n → F.Gv v →* Kˣ
  pol : ∀ g, r (F.star g) = Jmat K n * (r g)ᵀ * Jmat K n
  diag : ∀ i j, i ≠ j → (r F.g0 - r (F.star F.g0)) i j = 0
  ord : ∀ v g, ordProd (fun s => r (F.loc0 v s)) (fun i s => (ψ v i s : K)) g = 0

def map (f : K →+* K') (ρ : NRep F n K) : NRep F n K' where
  r := (f.mapMatrix : Matrix (Fin n) (Fin n) K →+* _).toMonoidHom.comp ρ.r
  ψ v i := (Units.map f.toMonoidHom).comp (ρ.ψ v i)
  pol g := by
    show f.mapMatrix (ρ.r (F.star g)) = Jmat K' n * (f.mapMatrix (ρ.r g))ᵀ * Jmat K' n
    rw [ρ.pol, map_mul, map_mul]
    simp only [RingHom.mapMatrix_apply, Jmat_map, Matrix.transpose_map]
  diag i j h := by
    show f (ρ.r F.g0 i j) - f (ρ.r (F.star F.g0) i j) = 0
    rw [← map_sub]
    exact (congrArg f (ρ.diag i j h)).trans (map_zero f)
  ord v g := (ordProd_map f _ _ g).trans (by rw [ρ.ord, map_zero])

def restrict {O : Type} [CommRing O] [Algebra O K] (ρ : NRep F n K) (A : Subalgebra O K)
    (h1 : ∀ g i j, ρ.r g i j ∈ A) (h2 : ∀ v i g, (ρ.ψ v i g : K) ∈ A) : NRep F n A where
  r :=
    { toFun := fun g => Matrix.of fun i j => ⟨ρ.r g i j, h1 g i j⟩
      map_one' := by
        ext i j
        by_cases h : i = j <;> simp [h, Matrix.one_apply]
      map_mul' := fun g h => by
        ext i j
        simp [Matrix.mul_apply] }
  ψ v i :=
    { toFun := fun g => ⟨⟨ρ.ψ v i g, h2 v i g⟩, ⟨ρ.ψ v i g⁻¹, h2 v i g⁻¹⟩, Subtype.ext (by simp), Subtype.ext (by simp)⟩
      map_one' := by ext; simp
      map_mul' := fun g h => by ext; simp }
  pol g := by
    refine Matrix.map_injective (f := (A.val : A →+* K)) Subtype.val_injective ?_
    show _ = (A.val : A →+* K).mapMatrix _
    rw [map_mul, map_mul]
    simp only [RingHom.mapMatrix_apply, Jmat_map]
    exact ρ.pol g
  diag i j h := Subtype.ext (ρ.diag i j h)
  ord v g := by
    refine Matrix.map_injective (f := (A.val : A →+* K)) Subtype.val_injective ?_
    show (A.val : A →+* K).mapMatrix _ = (A.val : A →+* K).mapMatrix _
    rw [← ordProd_map, map_zero]
    exact ρ.ord v g

def Cont (I : Ideal K) (ρ : NRep F n K) : Prop :=
  (∀ i j, AdicCont I fun g => ρ.r g i j) ∧ ∀ v i, AdicCont I fun g => (ρ.ψ v i g : K)

def IsBar (f : k →+* K) (ρ : NRep F n K) : Prop :=
  (∀ g i, ρ.r g i i = f ((F.Ψ g ^ w n i : kˣ) : k)) ∧ ∀ v i g, (ρ.ψ v i g : K) = f ((F.Ψ (F.loc v g) ^ w n i : kˣ) : k)

def IsDiag (ρ : NRep F n K) : Prop := ∀ g i j, i ≠ j → ρ.r g i j = 0

def triP (ρ : NRep F n K) : Ideal K := Ideal.span {x | ∃ g i j, j < i ∧ x = ρ.r g i j}

def triM (ρ : NRep F n K) : Ideal K := Ideal.span {x | ∃ g i j, i < j ∧ x = ρ.r g i j}

def ee (ρ : NRep F n K) : Ideal K := Ideal.span {x | ∃ g i j, Odd (w n i - w n j) ∧ x = ρ.r g i j}

def OffTri (ρ : NRep F n K) (p : Ideal K) : Prop := ¬ ρ.triP ≤ p ∧ ¬ ρ.triM ≤ p

def TI (ρ : NRep F n K) (p : Ideal K) : Prop := ρ.OffTri p ∧ (n = 3 → ¬ ρ.ee ≤ p)

def TE (ρ : NRep F n K) (p : Ideal K) : Prop := ρ.OffTri p ∧ n = 3 ∧ ρ.ee ≤ p

def Coin (ρ : NRep F n K) (p : Ideal K) (v : F.V) (i j : Fin n) : Prop :=
  ∀ g ∈ F.Nv v, (ρ.ψ v i g : K) - ρ.ψ v j g ∈ p

def adicClosure {O R : Type} [CommRing O] [CommRing R] [IsLocalRing R] [Algebra O R] (S : Subalgebra O R) :
    Subalgebra O R where
  carrier := {x | ∀ N : ℕ, ∃ y ∈ S, x - y ∈ maximalIdeal R ^ N}
  mul_mem' {a b} ha hb N := by
    obtain ⟨x, hx, h1⟩ := ha N
    obtain ⟨y, hy, h2⟩ := hb N
    refine ⟨x * y, mul_mem hx hy, ?_⟩
    have : a * b - x * y = (a - x) * b + x * (b - y) := by ring
    rw [this]
    exact add_mem (Ideal.mul_mem_right _ _ h1) (Ideal.mul_mem_left _ _ h2)
  add_mem' {a b} ha hb N := by
    obtain ⟨x, hx, h1⟩ := ha N
    obtain ⟨y, hy, h2⟩ := hb N
    refine ⟨x + y, add_mem hx hy, ?_⟩
    rw [add_sub_add_comm]
    exact add_mem h1 h2
  algebraMap_mem' c N := ⟨_, S.algebraMap_mem c, by simp⟩

theorem le_adicClosure {O R : Type} [CommRing O] [CommRing R] [IsLocalRing R] [Algebra O R] (S : Subalgebra O R) :
    S ≤ adicClosure S := fun x hx _ => ⟨x, hx, by simp⟩

private theorem trdeg_le_one {R D ι : Type} [CommRing R] [CommRing D] [IsDomain D] [Algebra R D]
    [FaithfulSMul R D] (x : ι → D) (wt : ι → ℤ) (hgen : Algebra.adjoin R (Set.range x) = ⊤)
    (h0 : ∀ (i j : ι) (p q : ℕ), p * wt i + q * wt j = 0 → x i ^ p * x j ^ q ∈ Set.range (algebraMap R D))
    {a b : ι} (ha : 0 < wt a) (hb : wt b < 0) (ha0 : x a ≠ 0) (hb0 : x b ≠ 0) : Algebra.trdeg R D ≤ 1 := by
  set T : Subalgebra R D := Algebra.adjoin R {x a} with hT
  have haT : x a ∈ T := Algebra.subset_adjoin rfl
  have hR : ∀ {y}, y ∈ Set.range (algebraMap R D) → y ∈ T := by
    rintro _ ⟨r, rfl⟩; exact T.algebraMap_mem r
  obtain ⟨p, hp⟩ := Int.eq_ofNat_of_zero_le ha.le
  obtain ⟨q, hq⟩ := Int.eq_ofNat_of_zero_le (neg_nonneg.2 hb.le)
  have hp0 : 0 < p := by omega
  have hq0 : 0 < q := by omega
  have hq' : wt b = -q := by omega
  have key : ∀ i, IsAlgebraic T (x i) := by
    intro i
    rcases le_total (wt i) 0 with h | h
    · obtain ⟨m, hm⟩ := Int.eq_ofNat_of_zero_le (neg_nonneg.2 h)
      have hm' : wt i = -m := by omega
      exact alg_of_rel T hp0 (pow_mem haT m) (hR (h0 i a p m (by rw [hm', hp]; ring))) (pow_ne_zero _ ha0) rfl
    · obtain ⟨m, hm⟩ := Int.eq_ofNat_of_zero_le h
      have h1 := hR (h0 i b q m (by rw [hm, hq']; ring))
      have h2 := hR (h0 b a p q (by rw [hp, hq']; ring))
      refine alg_of_rel T (mul_pos hq0 hp0) (pow_mem h2 m) (mul_mem (pow_mem h1 p) (pow_mem haT (q * m)))

    | add y z _ _ h1 h2 => exact h1.add h2

  simpa using Algebra.IsAlgebraic.trdeg_le_cardinalMk R ({x a} : Set D)

def NicePrimeEx : Prop :=
  ∀ (R : Type) [CommRing R] [IsNoetherianRing R] [IsLocalRing R] (ϖ : R), ϖ ∈ maximalIdeal R →
    ∀ (N : ℕ), 1 ≤ N → ∀ (ι : Type) [Finite ι] (a : ι → Ideal R), (∀ i, qdim (a i) < (N : ℕ)) →
    ∀ I : Ideal R, ((N + 1 : ℕ) : WithBot ℕ∞) ≤ qdim I →

structure LocData (R : Type) [CommRing R] where
  D1 : ℕ
  TI : Set (PrimeSpectrum R)
  TE : Set (PrimeSpectrum R)
  Good : Set (PrimeSpectrum R)
  Nice : Set (PrimeSpectrum R)
  Thick : Set (PrimeSpectrum R)
  sharp : Ideal R

variable {R : Type} [CommRing R] (L : LocData R)

def Comp (P : PrimeSpectrum R) : Prop := P.asIdeal ∈ L.sharp.minimalPrimes

def Adj (P Q : PrimeSpectrum R) : Prop :=
  L.Comp P ∧ L.Comp Q ∧ (L.D1 : WithBot ℕ∞) ≤ qdim (P.asIdeal ⊔ Q.asIdeal)

def Link (p q : PrimeSpectrum R) : Prop :=
  p ∈ L.Nice ∧ p ∈ L.TE ∧ q ∈ L.Nice ∧ q ∈ L.TE ∧ ∃ e ∈ L.Thick, L.Comp e ∧ e ≤ p ∧ e ≤ q

def P1 : Prop := ∀ p ∈ L.Nice, p ∈ L.TI → p ∈ L.Good → ∀ P ≤ p, P ∈ L.Good

def flipTwoThree : Prop := ∀ p ∈ L.Nice, p ∈ L.TE → ∀ q, Relation.ReflTransGen L.Link p q →
  (∃ P ≤ p, P ∈ L.TI ∧ P ∈ L.Good) → ∀ P' ≤ q, P' ∈ L.TI → P' ∈ L.Good

structure Ok : Prop where
  good_up : ∀ P Q, P ≤ Q → P ∈ L.Good → Q ∈ L.Good
  ti_down : ∀ P Q, P ≤ Q → Q ∈ L.TI → P ∈ L.TI
  nice_type : ∀ p ∈ L.Nice, p ∈ L.TI ∨ p ∈ L.TE
  nice_good : ∀ I ∈ L.Good, (L.D1 : WithBot ℕ∞) ≤ qdim I.asIdeal → ∃ p ∈ L.Nice, I ≤ p
  nice_thick : ∀ e ∈ L.Thick, L.Comp e → ∀ I, e ≤ I → (L.D1 : WithBot ℕ∞) ≤ qdim I.asIdeal →
    ∃ p ∈ L.Nice, p ∈ L.TE ∧ I ≤ p
  comp : ∀ P ∈ L.TI, P.asIdeal ∈ minimalPrimes R → L.Comp P
  chain : ∀ P Q, L.Comp P → L.Comp Q → Relation.ReflTransGen L.Adj P Q
  dich : ∀ P, L.Comp P → P ∈ L.TI ∨ P ∈ L.Thick
  p1 : L.P1
  p23 : L.flipTwoThree

def Concl : Prop := ∀ I ∈ L.Good, I ∈ L.TI → (L.D1 : WithBot ℕ∞) ≤ qdim I.asIdeal →
  ∀ P ∈ L.TI, P.asIdeal ∈ minimalPrimes R → P ∈ L.Good

  ∀ (x : Ideal S.A) (L : S.Loc x) (p : Ideal L.R), Nice L.ρ p → L.ρ.TI p → M ≤ L.π p →

  ∀ (x : Ideal S.A) (L : S.Loc x) (p q : Ideal L.R), Nice L.ρ p → L.ρ.TE p → Relation.ReflTransGen (ThickLink L.ρ) p q →

  ∃ (ξ : Ideal S.A) (Z : Ideal S.B), ξ.IsPrime ∧ S.u.TI ξ ∧ Z.IsPrime ∧ M ≤ Z ∧ Z ≤ S.π ξ ∧

    (h0 : 𝒜 0 = 1) (hinj : Function.Injective (algebraMap B A)) (p : Ideal B) [p.IsPrime] :
    ∃ η : Ideal A, η.IsPrime ∧ η.IsHomogeneous 𝒜 ∧ η.comap (algebraMap B A) = p := by
  classical
  have key : ∀ x ∈ Ideal.map (algebraMap B A) p, ∃ c ∈ p, GradedAlgebra.proj 𝒜 0 x = algebraMap B A c := by
    intro x hx
    have hx' : x ∈ (p • (⊤ : Submodule B A)) := by
      rw [Ideal.smul_top_eq_map]
      exact hx
    refine Submodule.smul_induction_on (p := fun x => ∃ c ∈ p, GradedAlgebra.proj 𝒜 0 x = algebraMap B A c)
      hx' ?_ ?_
    · intro b hb y _
      have hy : GradedAlgebra.proj 𝒜 0 y ∈ 𝒜 0 := by
        rw [GradedAlgebra.proj_apply]
        exact SetLike.coe_mem _
      rw [h0] at hy
      obtain ⟨c, hc⟩ := Submodule.mem_one.mp hy
      refine ⟨b * c, p.mul_mem_right c hb, ?_⟩
      rw [LinearMap.map_smul, ← hc, Algebra.smul_def, map_mul]
    · intro x y hx hy
      obtain ⟨c, hc, hxc⟩ := hx
      obtain ⟨d, hd, hyd⟩ := hy
      exact ⟨c + d, p.add_mem hc hd, by rw [map_add, hxc, hyd, map_add]⟩
  have hdisj : Disjoint ((Ideal.map (algebraMap B A) p : Ideal A) : Set A)
      (Algebra.algebraMapSubmonoid A p.primeCompl : Set A) := by
    rw [Set.disjoint_left]
    rintro x hxJ ⟨s, hs, rfl⟩
    obtain ⟨c, hc, he⟩ := key _ hxJ
    rw [GradedAlgebra.proj_apply, DirectSum.decompose_of_mem_same 𝒜 (SetLike.algebraMap_mem_graded 𝒜 s)] at he
    exact hs (by rw [hinj he]; exact hc)
  obtain ⟨P, hP, hJP, hPd⟩ := Ideal.exists_le_prime_disjoint _ _ hdisj
  refine ⟨(P.homogeneousCore 𝒜).toIdeal, Ideal.IsPrime.homogeneousCore (𝒜 := 𝒜) hP,
    (P.homogeneousCore 𝒜).isHomogeneous, ?_⟩
  apply le_antisymm
  · intro b hb
    rw [Ideal.mem_comap] at hb
    have hbP : algebraMap B A b ∈ P := Ideal.toIdeal_homogeneousCore_le 𝒜 P hb
    by_contra hnb
    exact Set.disjoint_left.mp hPd hbP ⟨b, hnb, rfl⟩
  · intro b hb
    rw [Ideal.mem_comap]
    exact Ideal.mem_homogeneousCore_of_homogeneous_of_mem ⟨0, SetLike.algebraMap_mem_graded 𝒜 b⟩
      (hJP (Ideal.mem_map_of_mem _ hb))

private theorem one_le_td {B A : Type} [CommRing B] [CommRing A] [Algebra B A] (𝒜 : ℤ → Submodule B A)
    [GradedAlgebra 𝒜] (η : Ideal A) (hη : η.IsPrime) (hh : η.IsHomogeneous 𝒜) {a : A} {α : ℤ} (ha : a ∈ 𝒜 α)
    (hα : α ≠ 0) (ha' : a ∉ η) : 1 ≤ Algebra.trdeg (B ⧸ η.comap (algebraMap B A)) (A ⧸ η) := by
  classical
  have key : ∀ Q : Polynomial B, Polynomial.aeval a Q ∈ η → ∀ k, Q.coeff k ∈ η.comap (algebraMap B A) := by
    intro Q hQ k
    have hn : Q.natDegree < Q.natDegree + k + 1 := by omega
    rw [Polynomial.aeval_eq_sum_range' hn] at hQ
    have hmem : ∀ i : ℕ, Q.coeff i • a ^ i ∈ 𝒜 (i • α) := fun i =>
      Submodule.smul_mem _ _ (SetLike.pow_mem_graded i ha)
    have h2 : GradedAlgebra.proj 𝒜 (k • α)
        (∑ i ∈ Finset.range (Q.natDegree + k + 1), Q.coeff i • a ^ i) ∈ η := by
      rw [GradedAlgebra.proj_apply]
      exact (Ideal.IsHomogeneous.mem_iff 𝒜 hh).1 hQ _
    rw [map_sum, Finset.sum_eq_single k] at h2
    · rw [GradedAlgebra.proj_apply, DirectSum.decompose_of_mem_same 𝒜 (hmem k), Algebra.smul_def] at h2
      rcases hη.mem_or_mem h2 with h | h
      · exact h
      · exact absurd (hη.mem_of_pow_mem _ h) ha'
    · intro i _ hik
      rw [GradedAlgebra.proj_apply, DirectSum.decompose_of_mem_ne 𝒜 (hmem i)]
      intro he
      apply hik
      have he' : (i : ℤ) * α = (k : ℤ) * α := by simpa [nsmul_eq_mul] using he
      exact_mod_cast mul_right_cancel₀ hα he'
    · intro hk
      exact absurd (Finset.mem_range.2 (by omega)) hk
  have htr : Transcendental (B ⧸ η.comap (algebraMap B A)) (Ideal.Quotient.mk η a) := by
    rw [transcendental_iff]
    intro P hP
    obtain ⟨Q, rfl⟩ := Polynomial.map_surjective (Ideal.Quotient.mk (η.comap (algebraMap B A)))
      Ideal.Quotient.mk_surjective P
    have e : Polynomial.aeval (Ideal.Quotient.mk η a) (Q.map (Ideal.Quotient.mk (η.comap (algebraMap B A))))
        = Ideal.Quotient.mk η (Polynomial.aeval a Q) := by
      have e1 := Polynomial.aeval_map_algebraMap (B ⧸ η.comap (algebraMap B A)) (Ideal.Quotient.mk η a) Q
      rw [Ideal.Quotient.algebraMap_eq] at e1
      rw [e1, ← Ideal.Quotient.mkₐ_eq_mk B η, Polynomial.aeval_algHom_apply]
    rw [e, Ideal.Quotient.eq_zero_iff_mem] at hP
    ext k
    rw [Polynomial.coeff_map, Polynomial.coeff_zero, Ideal.Quotient.eq_zero_iff_mem]
    exact key Q hP k
  haveI : Algebra.Transcendental (B ⧸ η.comap (algebraMap B A)) (A ⧸ η) := ⟨⟨_, htr⟩⟩
  exact Cardinal.one_le_iff_pos.2 (trdeg_pos _ _)

private theorem pair {B A : Type} [CommRing B] [CommRing A] [Algebra B A] (𝒜 : ℤ → Submodule B A)
    [SetLike.GradedMonoid 𝒜] (h0 : 𝒜 0 = 1) {ξ η : Ideal A} (hξ : ξ.IsPrime)
    (hle : η.comap (algebraMap B A) ≤ ξ.comap (algebraMap B A)) {a b : A} {α β : ℤ} (ha : a ∈ 𝒜 α) (hb : b ∈ 𝒜 β)
    (hα : 0 < α) (hβ : β < 0) (ha' : a ∉ ξ) (hb' : b ∉ ξ) : a ∉ η ∧ b ∉ η := by
  obtain ⟨m, hm⟩ : ∃ m : ℕ, (m : ℤ) = α := ⟨α.toNat, Int.toNat_of_nonneg hα.le⟩
  obtain ⟨k, hk⟩ : ∃ k : ℕ, (k : ℤ) = -β := ⟨(-β).toNat, Int.toNat_of_nonneg (by omega)⟩
  have hm0 : 0 < m := by omega
  have hk0 : 0 < k := by omega
  have hmem : a ^ k * b ^ m ∈ 𝒜 0 := by
    have h1 := SetLike.mul_mem_graded (SetLike.pow_mem_graded k ha) (SetLike.pow_mem_graded m hb)
    have he : k • α + m • β = 0 := by
      rw [nsmul_eq_mul, nsmul_eq_mul, hk, hm]
      ring
    rwa [he] at h1
  rw [h0] at hmem
  obtain ⟨x, hx⟩ := Submodule.mem_one.mp hmem
  have hc : a ^ k * b ^ m ∉ ξ := by
    intro hin
    rcases hξ.mem_or_mem hin with h | h
    · exact ha' (hξ.mem_of_pow_mem _ h)
    · exact hb' (hξ.mem_of_pow_mem _ h)
  have hxξ : x ∉ ξ.comap (algebraMap B A) := by
    rw [Ideal.mem_comap, hx]
    exact hc
  have hxη : x ∉ η.comap (algebraMap B A) := fun hin => hxξ (hle hin)
  rw [Ideal.mem_comap, hx] at hxη
  constructor
  · intro hin
    exact hxη (Ideal.mul_mem_right _ _ (Ideal.pow_mem_of_mem _ hin _ hk0))
  · intro hin
    exact hxη (Ideal.mul_mem_left _ _ (Ideal.pow_mem_of_mem _ hin _ hm0))

private theorem w_lt {n : ℕ} {i j : Fin n} (h : i < j) : w n j < w n i := by
  have : (i : ℕ) < j := h
  unfold w
  split_ifs <;> omega

theorem UpperTheory.V.Fr.frobenius_corner_lift_two_to_three :
    ∀ {l : ℕ} [Fact l.Prime] {O : Type} [CommRing O] [IsDomain O] [IsDiscreteValuationRing O] [IsCoef l O] {F : Frame l (ResidueField O)} {C₀ : ℕ} {n : ℕ}, F.Ok C₀ → (n = 2 ∨ n = 3) → ∀ S : Space O F n, ∀ (p : Ideal S.B) (ξ : Ideal S.A), p.IsPrime → ξ.IsPrime → S.u.TI ξ → p ≤ S.π ξ → ∃ η : Ideal S.A, η.IsPrime ∧ S.IsHom η ∧ S.π η = p ∧ S.u.TI η ∧ S.td η = 1 := by
  intro l _ O _ _ _ _ F C₀ n hF hn S p ξ hp hξ hTI hle
  obtain ⟨hint, hgm, h0, -, -, -, hr⟩ := gr_A hF hn S
  letI : GradedAlgebra S.gr := { hgm, hint.chooseDecomposition with }
  have hinj : Function.Injective (algebraMap S.B S.A) := by
    intro x y hxy
    apply Subtype.ext
    exact congrArg (fun z : S.A => (z : S.R)) hxy
  obtain ⟨η, hηp, hηh, hηc⟩ := ex_core S.gr h0 hinj p
  have hP := hTI.1.1
  have hM := hTI.1.2
  simp only [NRep.triP, NRep.triM, Ideal.span_le, Set.not_subset] at hP hM
  obtain ⟨_, ⟨g, i, j, hji, rfl⟩, hb⟩ := hP
  obtain ⟨_, ⟨g', i', j', hij', rfl⟩, ha⟩ := hM
  have hbd : w n i - w n j < 0 := by have := w_lt hji; omega
  have had : 0 < w n i' - w n j' := by have := w_lt hij'; omega
  have hle' : η.comap (algebraMap S.B S.A) ≤ ξ.comap (algebraMap S.B S.A) := by
    rw [hηc]; exact hle
  have hab := pair S.gr h0 hξ hle' (hr g' i' j') (hr g i j) had hbd ha hb
  refine ⟨η, hηp, ?_, hηc, ⟨⟨?_, ?_⟩, ?_⟩, ?_⟩
  · apply le_antisymm
    · have e := (Ideal.IsHomogeneous.iff_eq S.gr η).mp hηh
      calc η = _ := e.symm
        _ ≤ _ := Ideal.span_mono ?_
      rintro _ ⟨⟨x, hxh⟩, hx, rfl⟩
      exact ⟨hx, hxh⟩
    · exact Ideal.span_le.2 fun x hx => hx.1
  · intro h
    exact hab.2 (h (Ideal.subset_span ⟨g, i, j, hji, rfl⟩))
  · intro h
    exact hab.1 (h (Ideal.subset_span ⟨g', i', j', hij', rfl⟩))
  · intro h3 h
    have hE := hTI.2 h3
    simp only [NRep.ee, Ideal.span_le, Set.not_subset] at hE
    obtain ⟨_, ⟨g'', i'', j'', hodd, rfl⟩, hc⟩ := hE
    have hne : w n i'' - w n j'' ≠ 0 := by
      rintro h0'
      rw [h0'] at hodd
      exact (Int.not_odd_iff_even.2 ⟨0, by norm_num⟩) hodd
    have hcη : S.u.r g'' i'' j'' ∉ η := by
      rcases lt_or_gt_of_ne hne with hneg | hpos
      · exact (pair S.gr h0 hξ hle' (hr g' i' j') (hr g'' i'' j'') had hneg ha hc).2
      · exact (pair S.gr h0 hξ hle' (hr g'' i'' j'') (hr g i j) hpos hbd hc hb).1
    exact hcη (h (Ideal.subset_span ⟨g'', i'', j'', hodd, rfl⟩))
  · exact le_antisymm (fibre hF hn S η hηp ⟨fun h => hab.2 (h (Ideal.subset_span ⟨g, i, j, hji, rfl⟩)),
      fun h => hab.1 (h (Ideal.subset_span ⟨g', i', j', hij', rfl⟩))⟩)
      (one_le_td S.gr η hηp hηh (hr g' i' j') had.ne' hab.1)
